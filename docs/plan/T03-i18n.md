```ts
import { describe, expect, it } from 'vitest'
import en from '../../messages/en.json'
import es from '../../messages/es.json'
import fr from '../../messages/fr.json'

function keys(obj: Record<string, unknown>, prefix = ''): string[] {
  return Object.entries(obj).flatMap(([k, v]) =>
    v && typeof v === 'object' ? keys(v as Record<string, unknown>, `${prefix}${k}.`) : [`${prefix}${k}`],
  )
}

describe('messages', () => {
  const esKeys = keys(es).sort()
  it('EN has exactly the ES keys', () => expect(keys(en).sort()).toEqual(esKeys))
  it('FR has exactly the ES keys', () => expect(keys(fr).sort()).toEqual(esKeys))
  it('no empty strings', () => {
    for (const dict of [es, en, fr]) {
      const flat = JSON.stringify(dict)
      expect(flat).not.toContain('""')
    }
  })
})
```

`src/i18n/paths.ts`:

```ts
import { type Locale, locales } from './locales'
import { getPathname } from './navigation'
import type { AppPathname } from './routing'

export function localizedPaths(pathname: AppPathname, slugs?: Record<Locale, string>): Record<Locale, string> {
  return Object.fromEntries(
    locales.map((locale) => {
      const href = slugs ? { pathname, params: { slug: slugs[locale] } } : pathname
      // next-intl types require a literal pathname/params pair; the runtime accepts this shape.
      return [locale, getPathname({ locale, href: href as never })]
    }),
  ) as Record<Locale, string>
}
```

`src/i18n/request.ts`:

```ts
import { hasLocale } from 'next-intl'
import { getRequestConfig } from 'next-intl/server'
import { routing } from './routing'

export default getRequestConfig(async ({ requestLocale }) => {
  const requested = await requestLocale
  const locale = hasLocale(routing.locales, requested) ? requested : routing.defaultLocale
  return {
    locale,
    messages: (await import(`../../messages/${locale}.json`)).default,
  }
})
```

- [ ] **Step 6: Alternates context + language switcher**

`src/components/layout/alternates-context.tsx`:

```tsx
'use client'
import { createContext, type ReactNode, useContext } from 'react'
import type { Locale } from '@/i18n/locales'

type Alternates = Record<Locale, string> | null
const AlternatesContext = createContext<Alternates>(null)

export function AlternatesProvider({ value, children }: { value: Record<Locale, string>; children: ReactNode }) {
  return <AlternatesContext.Provider value={value}>{children}</AlternatesContext.Provider>
}

export function useAlternates(): Alternates {
  return useContext(AlternatesContext)
}
```

`src/components/layout/language-switcher.tsx`:

```tsx
'use client'
import { useLocale, useTranslations } from 'next-intl'
import { type Locale, localeLabels, locales } from '@/i18n/locales'
import { cn } from '@/lib/cn'
import { useAlternates } from './alternates-context'

const homePaths: Record<Locale, string> = { es: '/', en: '/en', fr: '/fr' }

export function LanguageSwitcher({ className }: { className?: string }) {
  const current = useLocale() as Locale
  const alternates = useAlternates()
  const t = useTranslations('nav')

  return (
    <nav aria-label={t('language')} className={cn('flex items-center gap-1 font-mono text-xs', className)}>
      {locales.map((locale) => (
        <a
          key={locale}
          href={alternates?.[locale] ?? homePaths[locale]}
          hrefLang={locale}
          lang={locale}
          aria-current={locale === current ? 'true' : undefined}
          title={localeLabels[locale]}
          className={cn(
            'rounded-full px-2.5 py-1 uppercase transition-colors',
            locale === current ? 'bg-amber text-bg-0' : 'text-ink-2 hover:text-ink-0',
          )}
        >
          {locale}
        </a>
      ))}
    </nav>
  )
}
```

> Plain `<a>` is intentional: switching language is a full navigation so the new `<html lang>` and fonts subset are rendered server-side.

`src/app/(frontend)/[locale]/layout.tsx`:

```tsx
import '../globals.css'
import { hasLocale, NextIntlClientProvider } from 'next-intl'
import { setRequestLocale } from 'next-intl/server'
import { notFound } from 'next/navigation'
import type { ReactNode } from 'react'
import { routing } from '@/i18n/routing'
import { fontVariables } from '@/lib/fonts'

export function generateStaticParams() {
  return routing.locales.map((locale) => ({ locale }))
}

export default async function LocaleLayout({
  children,
  params,
}: {
  children: ReactNode
  params: Promise<{ locale: string }>
}) {
  const { locale } = await params
  if (!hasLocale(routing.locales, locale)) notFound()
  setRequestLocale(locale)

  return (
    <html lang={locale} className={fontVariables} suppressHydrationWarning>
      <body className="grain bg-bg-0 text-ink-1">
        <NextIntlClientProvider>{children}</NextIntlClientProvider>
      </body>
    </html>
  )
}
```
